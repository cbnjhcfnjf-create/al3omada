-- ========================
-- DRAG FRAME LOGIC
-- ========================

local frameDragging = false
local frameDragStart, frameStartPos

local function setupDrag(dragSource)
    dragSource.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or
           input.UserInputType == Enum.UserInputType.MouseButton1 then
            frameDragging = true
            frameDragStart = input.Position
            frameStartPos = MainFrame.Position
        end
    end)
    dragSource.InputChanged:Connect(function(input)
        if frameDragging and (input.UserInputType == Enum.UserInputType.Touch or
           input.UserInputType == Enum.UserInputType.MouseMovement) then
            local delta = input.Position - frameDragStart
            MainFrame.Position = UDim2.new(
                frameStartPos.X.Scale,
                frameStartPos.X.Offset + delta.X,
                frameStartPos.Y.Scale,
                frameStartPos.Y.Offset + delta.Y
            )
        end
    end)
    dragSource.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or
           input.UserInputType == Enum.UserInputType.MouseButton1 then
            frameDragging = false
        end
    end)
end

local MIN_W, MIN_H = 250, 180
local resizing = false
local resizeCorner = nil
local resizeStart = nil
local resizeStartSize = nil
local resizeStartPos = nil

local CORNERS = {
    { name = "TL", ax = 0, ay = 0 },
    { name = "BL", ax = 0, ay = 1 },
    { name = "BR", ax = 1, ay = 1 },
}
local HANDLE_SIZE = 28

for _, c in ipairs(CORNERS) do
    local handle = Instance.new("TextButton")
    handle.Name = "ResizeHandle_" .. c.name
    handle.Size = UDim2.new(0, HANDLE_SIZE, 0, HANDLE_SIZE)
    handle.Position = UDim2.new(
        c.ax, c.ax == 1 and -HANDLE_SIZE or 0,
        c.ay, c.ay == 1 and -HANDLE_SIZE or 0
    )
    handle.BackgroundTransparency = 1
    handle.Text = ""
    handle.ZIndex = 10
    handle.Parent = MainFrame

    handle.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch and
           input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
        resizing = true
        resizeCorner = c
        resizeStart = input.Position
        resizeStartSize = Vector2.new(MainFrame.AbsoluteSize.X, MainFrame.AbsoluteSize.Y)
        resizeStartPos = Vector2.new(MainFrame.AbsolutePosition.X, MainFrame.AbsolutePosition.Y)
    end)
end

UserInputService.InputChanged:Connect(function(input)
    if not resizing then return end
    if input.UserInputType ~= Enum.UserInputType.Touch and
       input.UserInputType ~= Enum.UserInputType.MouseMovement then return end

    local delta = input.Position - resizeStart
    local c = resizeCorner
    local newW = c.ax == 1
        and math.max(MIN_W, resizeStartSize.X + delta.X)
        or  math.max(MIN_W, resizeStartSize.X - delta.X)
    local newH = c.ay == 1
        and math.max(MIN_H, resizeStartSize.Y + delta.Y)
        or  math.max(MIN_H, resizeStartSize.Y - delta.Y)
    local newX = c.ax == 0 and (resizeStartPos.X + (resizeStartSize.X - newW)) or resizeStartPos.X
    local newY = c.ay == 0 and (resizeStartPos.Y + (resizeStartSize.Y - newH)) or resizeStartPos.Y

    MainFrame.Size = UDim2.new(0, newW, 0, newH)
    MainFrame.Position = UDim2.new(0, newX, 0, newY)
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.Touch and
       input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
    resizing = false
    resizeCorner = nil
end)

            local hum2 = char2:FindFirstChildOfClass("Humanoid")
            if hum2 and hum2.Parent then hum2.PlatformStand = false end
        end)
    end

        updatePill(flyEnabled)

            -- FIX: Cleanup terus di sini, jangan bergantung pada RenderStepped

            return

        -- Cleanup stale instances
        local old1 = hrp:FindFirstChild("_FlyBV"); if old1 then old1:Destroy() end
        local old2 = hrp:FindFirstChild("_FlyBG"); if old2 then old2:Destroy() end

        local v3zero = Vector3.new(0,0,0)

                LocalPlayer.PlayerScripts
                    :WaitForChild("PlayerModule", 2)
                    :WaitForChild("ControlModule", 2)
            )
        end)

            -- Safety guard only (cleanup sudah dibuat dalam toggle handler)
            if not flyEnabled then return end

            local cam = workspace.CurrentCamera
            bg.CFrame  = cam.CoordinateFrame
            bv.Velocity = v3zero

            if ctrlOk and ctrl then
                -- Mobile (Delta): joystick via ControlModule
                -- FIX v8 #3: * 50 match IY mobilefly formula (iyflyspeed * 50)
                local dir = ctrl:GetMoveVector()
                if math.abs(dir.X) > 0.01 then
                    bv.Velocity = bv.Velocity + cam.CFrame.RightVector * (dir.X * flySpeed * 50)
                end
                if math.abs(dir.Z) > 0.01 then
                    bv.Velocity = bv.Velocity - cam.CFrame.LookVector * (dir.Z * flySpeed * 50)
                end
            else
                -- Desktop fallback: WASD + Space/Shift
                local d = v3zero
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then d = d + cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then d = d - cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then d = d - cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then d = d + cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space)     then d = d + Vector3.yAxis end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then d = d - Vector3.yAxis end
                if d.Magnitude > 0 then bv.Velocity = d.Unit * (flySpeed * 50) end
            end
        end)
    end)

        return {en=flyEnabled, sp=flySpeed, mn=flySliderMin, mx=flySliderMax}

            if not flyEnabled then flyCleanup() end
        end
    end

    local SPF = 5 * 2 * math.pi
    local vsP, vsV = Vector3.new(), Vector3.new()
    local psP, psV = Vector2.new(), Vector2.new()

    local function springV3(dt, goal)
        local offset = goal - vsP
        local decay  = math.exp(-SPF * dt)
        vsP = goal + (vsV * dt - offset * (SPF * dt + 1)) * decay
        vsV = (SPF * dt * (offset * SPF - vsV) + vsV) * decay
        return vsP
    end

    local function springV2(dt, goal)
        local offset = goal - psP
        local decay  = math.exp(-SPF * dt)
        psP = goal + (psV * dt - offset * (SPF * dt + 1)) * decay
        psV = (SPF * dt * (offset * SPF - psV) + psV) * decay
        return psP
    end

    local fcSavedType, fcSavedFov, fcSavedCF, fcSavedFocus

    local function startFreecam()
        if fcRunning then return end
        fcRunning = true

        local cam = workspace.CurrentCamera
        fcSavedType  = cam.CameraType

        fcSavedCF    = cam.CFrame

        -- FIX v8 #2: Init position + rotation dari current camera
        -- cam.CFrame:ToOrientation() → (rx=pitch, ry=yaw, rz=roll)
        -- Ini preserve arah pandang asal → freecam start dari view yang sama
        fcCamPos = cam.CFrame.Position
        local rx, ry, _ = cam.CFrame:ToOrientation()
        fcCamRot = Vector2.new(rx, ry)

        fcMDelta = Vector2.new()
        vsP, vsV = Vector3.new(), Vector3.new()
        psP, psV = Vector2.new(), Vector2.new()
        for k in pairs(fcK) do fcK[k] = false end

        pcall(function() cam.CameraType = Enum.CameraType.Scriptable end)

        fcKDConn = UserInputService.InputBegan:Connect(function(inp, proc)
            if proc then return end
            local k = inp.KeyCode.Name
            if fcK[k] ~= nil then fcK[k] = true end
        end)
        fcKUConn = UserInputService.InputEnded:Connect(function(inp)
            local k = inp.KeyCode.Name
            if fcK[k] ~= nil then fcK[k] = false end
        end)
        fcMouseConn = UserInputService.InputChanged:Connect(function(inp)
            if not fcRunning then return end
            if inp.UserInputType == Enum.UserInputType.MouseMovement then
                fcMDelta = fcMDelta + Vector2.new(-inp.Delta.Y, -inp.Delta.X)
            end
        end)

        fcRSConn = RunService.RenderStepped:Connect(function(dt)
            if not fcRunning then return end

            local rawVel = Vector3.new(
                (fcK.D and 1 or 0) - (fcK.A and 1 or 0),
                (fcK.E and 1 or 0) - (fcK.Q and 1 or 0),
                (fcK.S and 1 or 0) - (fcK.W and 1 or 0)
            )
            local shift = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift)
            local vel   = springV3(dt, rawVel * (shift and 0.25 or 1))

            local rawPan = fcMDelta * (math.pi / 64)
            fcMDelta = Vector2.new()
            local pan = springV2(dt, rawPan)

            fcCamRot = fcCamRot + pan * Vector2.new(0.75, 1) * 8 * dt
            fcCamRot = Vector2.new(
                math.clamp(fcCamRot.X, -math.rad(90), math.rad(90)),
                fcCamRot.Y % (2 * math.pi)
            )

            local newCF = CFrame.new(fcCamPos)
                * CFrame.fromOrientation(fcCamRot.X, fcCamRot.Y, 0)
                * CFrame.new(vel * 64 * dt)
            fcCamPos = newCF.Position

            local c = workspace.CurrentCamera
            pcall(function()
                c.CFrame    = newCF
                c.Focus     = newCF * CFrame.new(0, 0, -10)
                c.FieldOfView = fcFov
            end)
        end)
    end

    local function stopFreecam()
        if not fcRunning then return end
        fcRunning = false

        if fcRSConn    then fcRSConn:Disconnect();    fcRSConn    = nil end
        if fcKDConn    then fcKDConn:Disconnect();    fcKDConn    = nil end
        if fcKUConn    then fcKUConn:Disconnect();    fcKUConn    = nil end
        if fcMouseConn then fcMouseConn:Disconnect(); fcMouseConn = nil end

        for k in pairs(fcK) do fcK[k] = false end
        fcMDelta = Vector2.new()

        pcall(function()
            local cam = workspace.CurrentCamera
            cam.CameraType  = fcSavedType  or Enum.CameraType.Custom
            cam.FieldOfView = fcSavedFov   or 70
            if fcSavedCF    then cam.CFrame = fcSavedCF    end
            if fcSavedFocus then cam.Focus  = fcSavedFocus end
        end)
    end

    fcToggle.MouseButton1Click:Connect(function()
        if fcRunning then
            stopFreecam()
        else
            startFreecam()
        end
        updateFcPill(fcRunning)
    end)

            updateFcPill(fcRunning)
        end
    end

        if flEnabled then return end
        flEnabled = true

        local cam   = workspace.CurrentCamera
        flSavedType = cam.CameraType
        flSavedFov  = cam.FieldOfView

        -- Snapshot EXACT current camera position + orientation
        -- Position stays fixed wherever camera was (zoomed out, zoomed in, etc.)
        flCamPos = cam.CFrame.Position
        local rx, ry, _ = cam.CFrame:ToOrientation()
        flPitch  = rx
        flYaw    = ry
        flMDelta = Vector2.new()

        pcall(function() cam.CameraType = Enum.CameraType.Scriptable end)

        -- Accumulate mouse delta (Desktop: MouseMovement, Mobile: Touch)
        flMouseConn = UserInputService.InputChanged:Connect(function(inp)
            if not flEnabled then return end
            if inp.UserInputType == Enum.UserInputType.MouseMovement or
               inp.UserInputType == Enum.UserInputType.Touch then
                flMDelta = flMDelta + Vector2.new(-inp.Delta.Y, -inp.Delta.X)
            end
        end)

        -- RenderStepped: update rotation ONLY, position stays fixed
        flRSConn = RunService.RenderStepped:Connect(function()
            if not flEnabled then return end

            -- sensitivity ~0.006 rad/px (same feel as freecam pan)
            local sens = 0.006
            flPitch  = math.clamp(flPitch + flMDelta.X * sens, -math.rad(89), math.rad(89))
            flYaw    = (flYaw + flMDelta.Y * sens) % (2 * math.pi)
            flMDelta = Vector2.new()

            -- flCamPos stays fixed → no movement, only rotation
            pcall(function()
                local newCF = CFrame.new(flCamPos) * CFrame.fromOrientation(flPitch, flYaw, 0)
                workspace.CurrentCamera.CFrame      = newCF
                workspace.CurrentCamera.Focus       = newCF * CFrame.new(0, 0, -10)
                workspace.CurrentCamera.FieldOfView = flSavedFov or 70
            end)
        end)
    end

    local function stopFreeLook()
        if not flEnabled then return end
        flEnabled = false

        if flRSConn    then flRSConn:Disconnect();    flRSConn    = nil end
        if flMouseConn then flMouseConn:Disconnect(); flMouseConn = nil end
        flMDelta = Vector2.new()

        -- Restore camera to normal
        pcall(function()
            local cam       = workspace.CurrentCamera
            cam.CameraType  = flSavedType or Enum.CameraType.Custom
            cam.FieldOfView = flSavedFov  or 70
        end)
    end

    -- ── Handlekill logic ─────────────────────────────────────────
    local function hkTryKill()
        if not hkEnabled or #hkTargets == 0 then return end
        local myChar = LocalPlayer.Character
        local myHrp  = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if not myHrp then return end
        for _, target in ipairs(hkTargets) do
            local tChar = target.Character
            local tHrp  = tChar and tChar:FindFirstChild("HumanoidRootPart")
            if tHrp and (myHrp.Position - tHrp.Position).Magnitude <= hkRadius then
                local tHum = tChar:FindFirstChildOfClass("Humanoid")
                if tHum then
                    pcall(function() tHum:ChangeState(Enum.HumanoidStateType.Dead) end)
                end
            end
        end
    end

    local function hkListenChar(char)
        -- Disconnect any old tool connections
        if hkChildConn then hkChildConn:Disconnect(); hkChildConn = nil end
        if hkToolConn  then hkToolConn:Disconnect();  hkToolConn  = nil end
        -- Connect to any already-equipped tool
        local existTool = char:FindFirstChildOfClass("Tool")
        if existTool then
            hkToolConn = existTool.Activated:Connect(hkTryKill)
        end
        -- Watch for new tools being equipped
        hkChildConn = char.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then
                if hkToolConn then hkToolConn:Disconnect() end
                hkToolConn = child.Activated:Connect(hkTryKill)
            end
        end)
    end

    local function startHandleKill()
        hkEnabled = true
        updateHkPill(true)
        local char = LocalPlayer.Character
        if char then hkListenChar(char) end
        hkConn = LocalPlayer.CharacterAdded:Connect(function(c)
            hkListenChar(c)
        end)
    end

    local function stopHandleKill()
        hkEnabled = false
        updateHkPill(false)
        if hkConn      then hkConn:Disconnect();      hkConn      = nil end
        if hkChildConn then hkChildConn:Disconnect();  hkChildConn = nil end
        if hkToolConn  then hkToolConn:Disconnect();   hkToolConn  = nil end
    end

    hkToggleHitbox.MouseButton1Click:Connect(function()
        if hkEnabled then stopHandleKill() else startHandleKill() end
    end)

        local ps = hkPopup.AbsoluteSize

           tp.Y >= pa.Y and tp.Y <= pa.Y + ps.Y then return end

        local bs = hkPickerBtn.AbsoluteSize

        hkPickerArrow.Text = "▲"

    end)

    -- ── Freeze / Unfreeze logic ──────────────────────────────────
    -- Snapshot CFrame for one target into fpFrozenCF
    local function snapTarget(target)
        local char = target and target.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            fpFrozenCF[target] = hrp.CFrame
        end
    end

    -- Heartbeat loop: continuously lock all targets to their frozen CFrame
    -- NOTE: visual-only freeze (client-side), other player still moves on their end
    local function startFreezeLoop()
        if fpFreezeConn then fpFreezeConn:Disconnect() end
        fpFreezeConn = RunService.Heartbeat:Connect(function()
            for _, t in ipairs(fpTargets) do
                if t ~= LocalPlayer then
                    local cf = fpFrozenCF[t]
                    if cf then
                        pcall(function()
                            local char = t.Character
                            local hrp  = char and char:FindFirstChild("HumanoidRootPart")
                            if hrp then
                                hrp.CFrame = cf
                            end
                        end)
                    end
                end
            end
        end)
    end

    local function startFreeze()
        fpEnabled = true
        updateFpPill(true)
        -- Snapshot all current targets
        for _, t in ipairs(fpTargets) do
            snapTarget(t)
        end
        startFreezeLoop()
    end

    local function stopFreeze()
        fpEnabled = false
        updateFpPill(false)
        if fpFreezeConn then fpFreezeConn:Disconnect(); fpFreezeConn = nil end
        fpFrozenCF = {}
    end

                        end
                    end)
                end)

                end)
            end
        end

                    else

                    end
                end
                if nextTier and currentLevel >= nextTier.level then
                    RebirthRemote:FireServer()
                    task.wait(0.5)
                else
                    task.wait(1)
                end
            end
        end)

    local function isInsideHub_af(pos)
        local ok, result = pcall(function()
            local murSol = workspace.Structure.Stage0_HUB["Mur & Sol"]
            local cf, size = murSol:GetBoundingBox()
            local localPos = cf:PointToObjectSpace(pos)

                and math.abs(localPos.Z) < size.Z / 2
        end)
        if ok and result ~= nil then return result end
        return false
    end

    local function flyTo_af(targetPos)
        local char = LocalPlayer.Character
        if not char then return false end
        local root = char:FindFirstChild("HumanoidRootPart")
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not root or not humanoid then return false end

        for _, v in ipairs(root:GetChildren()) do
            if v:IsA("BodyGyro") or v:IsA("BodyVelocity") then v:Destroy() end
        end

        humanoid.PlatformStand = true
        local BG = Instance.new("BodyGyro")
        BG.P = 9e4; BG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        BG.CFrame = root.CFrame; BG.Parent = root
        local BV = Instance.new("BodyVelocity")
        BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        BV.Velocity = Vector3.new(0, 0, 0); BV.Parent = root

        local success = true
        while winFarmActive do
            if isInsideHub_af(root.Position) then success = false; break end
            local diff = targetPos - root.Position
            if diff.Magnitude < 3 then break end
            local dir = diff.Unit
            BV.Velocity = dir * (winFlySpeed * 50)
            BG.CFrame = CFrame.lookAt(root.Position, root.Position + dir)
            task.wait()
        end

        pcall(function() BV.Velocity = Vector3.new(0, 0, 0) end)
        pcall(function() BG:Destroy() end)
        pcall(function() BV:Destroy() end)
        pcall(function() humanoid.PlatformStand = false end)
        return success
    end

            end
        end)

    local cfgData     = {}    -- { [configName] = stateTable }
    local cfgAutoName = ""    -- config to auto-load on next execute
    local cfgSelName  = ""    -- currently selected config name in list

    -- ── File I/O ──────────────────────────────────────────────────
    local function cfgLoadFile()
        pcall(function()
            if not readfile then return end
            local raw = readfile(cfgFileName)
            if not raw or raw == "" then return end
            local dec = HttpService:JSONDecode(raw)
            if type(dec) == "table" then
                cfgData     = type(dec.configs)  == "table"  and dec.configs  or {}
                cfgAutoName = type(dec.autoload) == "string" and dec.autoload or ""
            end
        end)
    end

    local function cfgSaveFile()
        pcall(function()
            if not writefile then return end
            writefile(cfgFileName, HttpService:JSONEncode({
                configs  = cfgData,
                autoload = cfgAutoName,
            }))
        end)
    end

    cfgLoadFile()

    -- ── State helpers ─────────────────────────────────────────────
    local function cfgGetState()
        local t = {}
        for k, fn in pairs(cfgRead) do
            t[k] = fn()
        end
        return t
    end

    local function cfgSetState(state)
        if type(state) ~= "table" then return end
        for k, fn in pairs(cfgApply) do
            if state[k] ~= nil then
                pcall(fn, state[k])
            end
        end
    end

            cfgAutoName = ""       -- toggle off
        else
            cfgAutoName = cfgSelName
        end
        cfgSaveFile()
        renderCfgList()
    end)

    -- Delete: remove selected config
    cfgDeleteBtn.MouseButton1Click:Connect(function()
        if cfgSelName == "" or cfgData[cfgSelName] == nil then return end
        cfgData[cfgSelName] = nil
        if cfgAutoName == cfgSelName then cfgAutoName = "" end
        cfgSaveFile()
        cfgSelName = ""
        cfgNameBox.Text = ""
        renderCfgList()
    end)

    -- ── Re-render list when Tab3 (Save Config) becomes active ────
    Tab3.MouseButton1Click:Connect(function()
        renderCfgList()
    end)

    -- ── Auto-load on script execute ───────────────────────────────
    if cfgAutoName ~= "" and cfgData[cfgAutoName] ~= nil then
        task.defer(function()
            cfgSetState(cfgData[cfgAutoName])
        end)
    end

end -- end SAVE CONFIG

-- CheatHub LEFT recalc: Movement → Camera → Player → Misc
local function recalcCheatDropPositions()
    local tweenI   = TweenInfo.new(0.2, Enum.EasingStyle.Quart)
    local currentY = 5

    local movH = cheatDropState.movOpen and MOV_DROP_OPEN_H or DROP_CLOSED_H
    TweenService:Create(FlyWrapper, tweenI, {
        Position = UDim2.new(0,5,0,currentY), Size = UDim2.new(1,-10,0,movH)
    }):Play()
    currentY = currentY + movH + DROP_GAP

    local camH = cheatDropState.camOpen and CAM_DROP_OPEN_H or DROP_CLOSED_H
    TweenService:Create(CameraWrapper, tweenI, {
        Position = UDim2.new(0,5,0,currentY), Size = UDim2.new(1,-10,0,camH)
    }):Play()
    currentY = currentY + camH + DROP_GAP

    local playerH = cheatDropState.playerOpen and PLAYER_DROP_OPEN_H or DROP_CLOSED_H
    TweenService:Create(PlayerWrapper, tweenI, {
        Position = UDim2.new(0,5,0,currentY), Size = UDim2.new(1,-10,0,playerH)
    }):Play()
    currentY = currentY + playerH + DROP_GAP

    local miscH = cheatDropState.miscOpen and MISC_DROP_OPEN_H or DROP_CLOSED_H
    TweenService:Create(MiscWrapper, tweenI, {
        Position = UDim2.new(0,5,0,currentY), Size = UDim2.new(1,-10,0,miscH)
    }):Play()

    updateScrollState()
end

local function applySearch(query)
    local q = query:lower()

    -- CheatHub LEFT panel
    do
        local items = {
            { FlyWrapper,    "movement" },
            { CameraWrapper, "camera"   },
            { PlayerWrapper, "player"   },
            { MiscWrapper,   "misc"     },
        }
        local curY = 5
        for _, it in ipairs(items) do
            local show = q == "" or it[2]:find(q, 1, true) ~= nil
            it[1].Visible = show
            if show then
                it[1].Position = UDim2.new(0, 5, 0, curY)
                curY = curY + it[1].Size.Y.Offset + DROP_GAP
            end
        end
    end

    -- CheatHub RIGHT panel
    do
        local items = {
            { TeleportWrapper, "teleport" },
            { AntiWrapper,     "anti"     },
            { ServerWrapper,   "server"   },
        }
        local curY = 5
        for _, it in ipairs(items) do
            local show = q == "" or it[2]:find(q, 1, true) ~= nil
            it[1].Visible = show
            if show then
                it[1].Position = UDim2.new(0, 5, 0, curY)
                curY = curY + it[1].Size.Y.Offset + DROP_GAP
            end
        end
    end

    updateScrollState()
end

-- ============================================================
-- CLEANUP FUNCTION
-- Stop semua active farm threads + destroy GUI.
-- Dipanggil oleh: CloseBtn, atau execute baru (via _G guard).
-- ============================================================
local function cleanupAll()
    pcall(function() if _G.VORTX_stopTreadmill then _G.VORTX_stopTreadmill() end end)
    pcall(function() if _G.VORTX_stopRebirth   then _G.VORTX_stopRebirth()   end end)
    pcall(function() if _G.VORTX_stopWinFarm   then _G.VORTX_stopWinFarm()   end end)
    pcall(function() ScreenGui:Destroy() end)
    _G.UITemplate_Cleanup   = nil
    _G.VORTX_stopTreadmill  = nil
    _G.VORTX_stopRebirth    = nil
    _G.VORTX_stopWinFarm    = nil
end

-- Register ke _G — execute baru akan panggil ini dulu
_G.UITemplate_Cleanup = cleanupAll
