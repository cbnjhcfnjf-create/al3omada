local function Gradient(colors, rotation)
    local points = {}
    for position, color in pairs(colors) do
        if type(color) == "table" then
            points[position] = {
                Color = Color3.fromHex(color.Color),
                Transparency = color.Transparency or 0
            }
        else
            points[position] = { Color = Color3.fromHex(color), Transparency = 0 }
        end
    end
    return A:Gradient(points, { Rotation = rotation or 45 })
end

local CarControlEnabled = false
local CarSpeed = 60
local CarRotateSpeed = 3
local Acceleration = 0.2
local Deceleration = 0.05
local RotationSmoothness = 5
local RotateSpeedLimit = 0.05
local LockHeightEnabled = true

-- 🆕 Z/X
local VerticalSpeed = 15
local VerticalSmoothness = 8
local currentVerticalVelocity = 0
local heightOffset = 0

local currentSeat = nil
local currentPrimary = nil
local controlConnection = nil
local linearVelocity = nil
local alignOrientation = nil
local antiGravityForce = nil
local att1, att2, att3

local function StopCarControl()
    CarControlEnabled = false
    currentVerticalVelocity = 0
    heightOffset = 0
    
    if controlConnection then controlConnection:Disconnect() controlConnection = nil end
    if linearVelocity then linearVelocity:Destroy() linearVelocity = nil end
    if alignOrientation then alignOrientation:Destroy() alignOrientation = nil end
    if antiGravityForce then antiGravityForce:Destroy() antiGravityForce = nil end
    if att1 then att1:Destroy() att1 = nil end
    if att2 then att2:Destroy() att2 = nil end
    if att3 then att3:Destroy() att3 = nil end
    
    currentPrimary = nil
end

local function StartCarControl(seat)
    if not seat then return end
    StopCarControl()
    
    CarControlEnabled = true
    currentSeat = seat
    local primaryPart = seat.AssemblyRootPart or seat
    currentPrimary = primaryPart
    local lockedY = primaryPart.Position.Y
    
    primaryPart.CustomPhysicalProperties = PhysicalProperties.new(0.7, 0.3, 0.5, 1, 1)
    
    att1 = Instance.new("Attachment") att1.Parent = primaryPart
    att2 = Instance.new("Attachment") att2.Parent = primaryPart
    att3 = Instance.new("Attachment") att3.Parent = primaryPart
    
    alignOrientation = Instance.new("AlignOrientation")
    alignOrientation.Attachment0 = att1
    alignOrientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
    alignOrientation.MaxTorque = 500000
    alignOrientation.Responsiveness = 100
    alignOrientation.RigidityEnabled = true
    alignOrientation.PrimaryAxisOnly = false
    alignOrientation.CFrame = CFrame.new(primaryPart.Position) 
        * CFrame.Angles(0, math.atan2(-primaryPart.CFrame.LookVector.X, -primaryPart.CFrame.LookVector.Z), 0)
    alignOrientation.Parent = primaryPart
    
    linearVelocity = Instance.new("LinearVelocity")
    linearVelocity.Attachment0 = att2
    linearVelocity.MaxForce = 100000
    linearVelocity.VectorVelocity = Vector3.new(0, 0, 0)
    linearVelocity.ForceLimitMode = Enum.ForceLimitMode.PerAxis
    linearVelocity.MaxAxesForce = Vector3.new(100000, 0, 100000)
    linearVelocity.Parent = primaryPart
    
    antiGravityForce = Instance.new("VectorForce")
    antiGravityForce.Attachment0 = att3
    antiGravityForce.Force = Vector3.new(0, primaryPart.AssemblyMass * workspace.Gravity, 0)
    antiGravityForce.RelativeTo = Enum.ActuatorRelativeTo.World
    antiGravityForce.Parent = primaryPart
    
    local targetYaw = math.atan2(-primaryPart.CFrame.LookVector.X, -primaryPart.CFrame.LookVector.Z)
    local currentYaw = targetYaw
    local currentVelocity = Vector3.new(0, 0, 0)
    local flipCheckCooldown = 0
    
    currentVerticalVelocity = 0
    heightOffset = 0
    
    controlConnection = RunService.Heartbeat:Connect(function(dt)
        if not CarControlEnabled then return end
        if not primaryPart or not primaryPart.Parent then StopCarControl() return end
        if not currentSeat or not currentSeat.Parent then StopCarControl() return end
        
        -- 🔄 اللف
        local rotateInput = 0
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then rotateInput = rotateInput + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then rotateInput = rotateInput - 1 end
        
        if rotateInput ~= 0 then
            targetYaw = targetYaw + rotateInput * CarRotateSpeed * dt
        end
        
        local yawDiff = targetYaw - currentYaw
        while yawDiff > math.pi do yawDiff = yawDiff - math.pi * 2 end
        while yawDiff < -math.pi do yawDiff = yawDiff + math.pi * 2 end
        
        local yawStep = math.clamp(yawDiff * RotationSmoothness, -RotateSpeedLimit, RotateSpeedLimit)
        currentYaw = currentYaw + yawStep
        
        -- 🆕 Z/X
        local vInput = 0
        if UserInputService:IsKeyDown(Enum.KeyCode.Z) then vInput = vInput + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.X) then vInput = vInput - 1 end
        
        local targetVVel = vInput * VerticalSpeed
        currentVerticalVelocity = currentVerticalVelocity + 
            (targetVVel - currentVerticalVelocity) * math.min(VerticalSmoothness * dt, 1)
        
        if math.abs(currentVerticalVelocity) > 0.1 then
            heightOffset = heightOffset + currentVerticalVelocity * dt
        end
        
        -- تثبيت السيارة
        local pos = primaryPart.Position
        local baseY = LockHeightEnabled and (lockedY + heightOffset) or pos.Y
        local newCFrame = CFrame.new(pos.X, baseY, pos.Z) * CFrame.Angles(0, currentYaw, 0)
        alignOrientation.CFrame = newCFrame
        
        -- 🚗 الحركة
        local moveInput = 0
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveInput = moveInput + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveInput = moveInput - 1 end
        
        local forward = Vector3.new(newCFrame.LookVector.X, 0, newCFrame.LookVector.Z).Unit
        
        if moveInput ~= 0 then
            local targetVel = forward * moveInput * CarSpeed
            currentVelocity = currentVelocity:Lerp(targetVel, Acceleration)
        else
            currentVelocity = currentVelocity:Lerp(Vector3.new(0, 0, 0), Deceleration)
            if currentVelocity.Magnitude < 0.5 then
                currentVelocity = Vector3.new(0, 0, 0)
            end
        end
        
        if math.abs(rotateInput) > 0 then
            currentVelocity = currentVelocity * 0.85
        end
        
        linearVelocity.VectorVelocity = currentVelocity
        
        -- Anti-Flip
        flipCheckCooldown = flipCheckCooldown + dt
        if flipCheckCooldown >= 0.05 then
            flipCheckCooldown = 0
            local upVec = primaryPart.CFrame.UpVector
            if math.abs(upVec.Y) < 0.7 then
                primaryPart.CFrame = CFrame.new(pos.X, baseY, pos.Z) * CFrame.Angles(0, currentYaw, 0)
                primaryPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                primaryPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                currentVelocity = Vector3.new(0, 0, 0)
            end
        end
    end)
    
    A:Notify({ 
        Title = "🚗 سيارة", 
        Content = "التحكم شغال - Z فوق | X تحت", 
        Duration = 2 
    })
end

local function SetupCharacter(char)
    local humanoid = char:WaitForChild("Humanoid", 5)
    if not humanoid then return end
    
    humanoid.Seated:Connect(function(active, seat)
        if active and seat and seat:IsA("VehicleSeat") then
            local vehicle = seat:FindFirstAncestorOfClass("Model") or seat
            A:Notify({ 
                Title = "🚗", 
                Content = "ركبت: " .. (vehicle.Name or "سيارة"), 
                Duration = 2 
            })
            if CarControlEnabled then
                StartCarControl(seat)
            end
        else
            if controlConnection then StopCarControl() end
        end
    end)
end

local BuyConfig = {
    ItemName = "WormtecRegular",
    Amount = 10,
    Delay = 0.5,
    Enabled = false,
    EventName = "MopShopEvent",
}

local buyThread = nil
local buyCount = 0

local function getShopEvent()
    local ev = ReplicatedStorage:FindFirstChild(BuyConfig.EventName)
    if ev and ev:IsA("RemoteEvent") then
        return ev
    end
    return nil
end

local function StopAutoBuy()
    BuyConfig.Enabled = false
    if buyThread then
        pcall(function() task.cancel(buyThread) end)
        buyThread = nil
    end
    A:Notify({
        Title = "🛒 Auto Buy",
        Content = "تم الإيقاف | العمليات: " .. buyCount,
        Duration = 2
    })
end

local function StartAutoBuy()
    local ev = getShopEvent()
    if not ev then
        A:Notify({
            Title = "❌ خطأ",
            Content = "ما لقيت " .. BuyConfig.EventName,
            Duration = 3
        })
        return
    end
    
    BuyConfig.Enabled = true
    buyCount = 0
    
    buyThread = task.spawn(function()
        while BuyConfig.Enabled do
            local ok, err = pcall(function()
                ev:FireServer("BUY", BuyConfig.ItemName, BuyConfig.Amount)
            end)
            
            buyCount = buyCount + 1
            
            if not ok then
                warn("[AutoBuyer] Error:", err)
                BuyConfig.Enabled = false
                A:Notify({ Title = "❌ خطأ", Content = tostring(err), Duration = 3 })
                break
            end
            
            task.wait(BuyConfig.Delay)
        end
    end)
    
    A:Notify({
        Title = "🛒 Auto Buy",
        Content = "اشتغل | " .. BuyConfig.ItemName .. " x" .. BuyConfig.Amount,
        Duration = 2
    })
end

task.spawn(function()
    local sellEvent = ReplicatedStorage:WaitForChild("SellShopEvent", 10)
    if not sellEvent then
        warn("[AutoSell] SellShopEvent not found")
        return
    end
    
    local SellArguments = {"SellAll", "Sell All", "SellAllItems", "SellItems", "SellGold", "SellFish", "SellAllFish", "All"}
    
    while task.wait(1) do 
        if _G.AutoSellAll then
            pcall(function()
                sellEvent:FireServer()

-- ═══ دالة جلب السيارة اللي spawnتها ═══
local function FindSpawnedCar()
    -- نبحث في workspace عن أقرب سيارة للاعب
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then
        return nil
    end
    
    local myPos = char.HumanoidRootPart.Position
    local closest = nil
    local closestDist = math.huge
    
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and obj.PrimaryPart then
            -- نتحقق إن الاسم يحتوي على اسم سيارة
            for carName, _ in pairs(Cars) do
                if string.find(obj.Name, carName) then
                    local dist = (obj.PrimaryPart.Position - myPos).Magnitude
                    if dist < closestDist and dist < 500 then  -- ضمن 500 ستود
                        closest = obj
                        closestDist = dist
                    end
                end
            end
        end
    end
    
    return closest
end

-- ═══ دالة نقل السيارة قدامك ═══
local function MoveCarToMe(car)
    if not car or not car.PrimaryPart then return end
    
    local char = LocalPlayer.Character
    if not char then return end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    -- الموقع قدام اللاعب
    local targetPos = root.Position + (root.CFrame.LookVector * Config.Distance) + Vector3.new(0, Config.Height, 0)
    
    -- نستخدم PivotTo عشان ننقل الموديل كامل
    pcall(function()
        car:PivotTo(CFrame.new(targetPos))
    end)
end

        if SpawnedCar and SpawnedCar.Parent then
            MoveCarToMe(SpawnedCar)
        end
    end)
end

local function StartHold()
    if SnapConfig.HoldThread then
        pcall(function() task.cancel(SnapConfig.HoldThread) end)
        SnapConfig.HoldThread = nil
    end
    
    SnapConfig.HoldThread = task.spawn(function()
        while SnapConfig.IsDown do
            local char = player.Character
            if not char then break end
            
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then break end
            
            -- نثبته في مكانه كل فريم
            if SnapConfig.SavedY then
                local targetY = SnapConfig.SavedY - SnapConfig.DownDistance

                root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                
                -- نثبت الموقع
                root.CFrame = CFrame.new(root.Position.X, targetY, root.Position.Z)
            end
            
            RunService.Heartbeat:Wait()
        end
    end)
end

        Duration = 1.5
    })

-- ═══ Snap طلوع ═══
local function SnapUp()
    local char = player.Character
    if not char then return end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    -- نوقف التثبيت أول
    StopHold()
    SnapConfig.IsDown = false
    
    -- نكتشف السطح
    local rayOrigin = root.Position + Vector3.new(0, 5, 0)
    local rayDir    = Vector3.new(0, -1, 0) * 1000
    
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {char}
    
    local result = workspace:Raycast(rayOrigin, rayDir, params)
    
    local targetY
    if result then
        targetY = result.Position.Y + 3.5
    else
        targetY = (SnapConfig.SavedY or root.Position.Y) + SnapConfig.UpDistance
    end
    
    -- Snap فوري
    root.CFrame = CFrame.new(root.Position.X, targetY, root.Position.Z)
    root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    
    A:Notify({
        Title = "Snap Up",
        Content = "طلعت فوق السطح",
        Duration = 1.5
    })
end

local function EquipTool1()
    local char = player.Character
    if not char then return false end
    
    local backpack = player:FindFirstChild("Backpack")
    if not backpack then return false end
    
    -- نجيب كل الأدوات
    local tools = {}
    for _, item in ipairs(backpack:GetChildren()) do
        if item:IsA("Tool") then
            table.insert(tools, item)
        end
    end
    
    -- نرتب حسب الاسم
    table.sort(tools, function(a, b) return a.Name < b.Name end)
    
    if #tools == 0 then
        return false
    end
    
    -- نمسك الأداة الأولى
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if humanoid then
        humanoid:EquipTool(tools[1])
        return true
    end
    
    return false
end

local function GetRestaurantWater()
    local waterList = {}
    
    local interior = workspace:FindFirstChild("Interior")
    if not interior then return waterList end
    
    local puddles = interior:FindFirstChild("Puddles")

-- ═══ دالة الطيران فوق الماء ═══
local function FlyToWater(water)
    local char = player.Character
    if not char then return false end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    
    local waterPos
    if water:IsA("BasePart") then
        waterPos = water.Position
    elseif water:IsA("Model") and water.PrimaryPart then
        waterPos = water.PrimaryPart.Position
    else
        local part = water:FindFirstChildWhichIsA("BasePart")
        if part then waterPos = part.Position else return false end
    end
    
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    
    local targetPos = waterPos + Vector3.new(0, 3, 0)
    local reached = false
    local startTime = tick()
    
    flyConnection = RunService.Heartbeat:Connect(function(dt)
        if not char.Parent then
            flyConnection:Disconnect()
            return
        end
        
        local r = char:FindFirstChild("HumanoidRootPart")
        if not r then return end
        
        local currentPos = r.Position
        local diff = targetPos - currentPos
        local distance = diff.Magnitude
        
        if distance < 2 or tick() - startTime > 3 then
            r.CFrame = CFrame.new(targetPos)
            reached = true
            flyConnection:Disconnect()
            flyConnection = nil
            return
        end
        
        local direction = diff.Unit
        local step = math.min(FarmConfig.FlySpeed * dt, distance)
        r.CFrame = CFrame.new(currentPos + direction * step)
        r.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    end)
    
    local waitTime = 0
    while not reached and waitTime < 3.5 do
        task.wait(0.05)
        waitTime = waitTime + 0.05
    end
    
    return reached
end

-- ═══ دالة البقاء فوق الماء ═══
local function HoldAboveWater(water)
    local char = player.Character
    if not char then return end
    
    local waterPos
    if water:IsA("BasePart") then
        waterPos = water.Position
    elseif water:IsA("Model") and water.PrimaryPart then
        waterPos = water.PrimaryPart.Position
    else
        local part = water:FindFirstChildWhichIsA("BasePart")
        if part then waterPos = part.Position else return end
    end
    
    local targetPos = waterPos + Vector3.new(0, 3, 0)
    local holdEnd = tick() + FarmConfig.FlyDuration
    
    while tick() < holdEnd and FarmConfig.Enabled do
        local r = char:FindFirstChild("HumanoidRootPart")
        if not r then break end
        
        r.CFrame = CFrame.new(targetPos)
        r.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        r.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        
        RunService.Heartbeat:Wait()
    end
end

-- ═══ دالة تنظيف الماء ═══
local function CleanWater()
    local cleanEvent = getEvent(FarmConfig.CleanEvent)
    if not cleanEvent then
        return false, "ما لقيت الحدث"
    end
    
    local waterList = GetRestaurantWater()
    if #waterList == 0 then
        return false, "ما فيه ماء"
    end
    
    local cleaned = 0
    for _, water in ipairs(waterList) do
        pcall(function()
            cleanEvent:FireServer("Start", water)
            task.wait(0.05)
            cleanEvent:FireServer("Complete", water)
        end)
        
        cleaned = cleaned + 1
        
        if FarmConfig.GoToWater then
            local arrived = FlyToWater(water)
            if arrived then
                HoldAboveWater(water)
            end
        end
    end
    
    return true, cleaned
end

--  🎯 SERVER HOP - يدخلك سيرفر بعدد محدد

-- ═══ الإعدادات ═══
local HopConfig = {
    TargetCount = 1,       -- عدد اللاعبين المطلوب
    MaxPages    = 10,      -- كم صفحة يدور
    ExactMatch  = false,   -- true = بالضبط | false = أقل أو يساوي
    Enabled     = false,
}

local hopThread = nil

-- ═══ دالة جلب السيرفرات ═══
local function GetServers()
    local servers = {}
    local cursor = ""
    
    for page = 1, HopConfig.MaxPages do
        local url = string.format(
            "https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100&cursor=%s",
            game.PlaceId, cursor
        )
        
        local ok, response = pcall(function()
            return game:HttpGet(url)
        end)
        
        if not ok then
            warn("[ServerHop] فشل الاتصال")
            break
        end
        
        local ok2, data = pcall(function()
            return HttpService:JSONDecode(response)
        end)
        
        if not ok2 or not data or not data.data then
            break
        end
        
        for _, server in ipairs(data.data) do
            -- نتجاهل السيرفر الحالي
            if server.id ~= game.JobId then
                local playerCount = server.playing or 0
                local maxPlayers  = server.maxPlayers or 0
                
                -- نتحقق من العدد
                local match = false
                if HopConfig.ExactMatch then
                    match = (playerCount == HopConfig.TargetCount)
                else
                    match = (playerCount <= HopConfig.TargetCount)
                end
                
                -- نتأكد إن السيرفر مو ممتلئ
                if match and playerCount < maxPlayers then
                    table.insert(servers, {
                        id = server.id,
                        playing = playerCount,
                        maxPlayers = maxPlayers,
                        ping = server.ping or 0,
                    })
                end
            end
        end
        
        -- إذا لقينا سيرفرات كافية، نوقف
        if #servers >= 5 then
            break
        end
        
        cursor = data.nextPageCursor
        if not cursor or cursor == "" then
            break
        end
    end
    
    return servers
end

-- ═══ دالة الدخول لسيرفر ═══
local function JoinServer(serverId)
    local ok, err = pcall(function()
        TeleportService:TeleportToPlaceInstance(game.PlaceId, serverId, player)
    end)
    
    if not ok then
        warn("[ServerHop] فشل الدخول:", err)
    end
    
    return ok
end

-- ═══ الإيقاف ═══
local function StopHop()
    HopConfig.Enabled = false
    if hopThread then
        pcall(function() task.cancel(hopThread) end)
        hopThread = nil
    end
end

-- ═══ البدء ═══
local function StartHop()
    HopConfig.Enabled = true
    
    hopThread = task.spawn(function()
        while HopConfig.Enabled do
            print("[ServerHop] 🔍 جاري البحث عن سيرفر فيه " .. HopConfig.TargetCount .. " لاعب...")
            
            local servers = GetServers()
            
            if #servers == 0 then
                print("[ServerHop] ❌ ما لقيت سيرفر، نحاول مرة ثانية...")
                task.wait(3)
            else
                -- نختار سيرفر عشوائي من النتائج
                local chosen = servers[math.random(1, #servers)]
                
                print(string.format(
                    "[ServerHop] ✅ لقيت سيرفر: %d/%d لاعب - Job ID: %s",
                    chosen.playing, chosen.maxPlayers, chosen.id
                ))
                
                JoinServer(chosen.id)
                break
            end
        end
    end)
end

local FishSellTab = Window:Tab({ Title = "بيع الاسماك", Icon = "fish" })

-- ═══ الإعدادات ═══
local FishSellConfig = {
    EventName = "SellShopEvent",
    Delay     = 0.15,
    Enabled   = false,
}

-- ═══ الأسماك ═══
local FishList = {
    { Name = "Tuna",         DebugId = "1_4535164", Amount = 99 },
    { Name = "Dolphinfish",  DebugId = "1_4535169", Amount = 99 },
    { Name = "Coelacanth",   DebugId = "1_4535202", Amount = 99 },
    { Name = "NorthernPike", DebugId = "1_4537790", Amount = 99 },
    { Name = "Sailfish",     DebugId = "1_4544279", Amount = 99 },
    { Name = "Marlin",       DebugId = "1_4544527", Amount = 99 },
}

local FishEnabled = {}
for _, fish in ipairs(FishList) do
    FishEnabled[fish.Name] = true
end

local sellThread = nil
local sellCount  = 0

-- ═══ دالة جلب الحدث ═══
local function getSellEvent()
    local ev = ReplicatedStorage:FindFirstChild(FishSellConfig.EventName)
    if ev and ev:IsA("RemoteEvent") then
        return ev
    end
    return nil
end

-- ═══ دالة تجميع كل الأسماك ═══
local function CollectAllFish()
    local collected = {}
    
    local nilInstances = nil
    if getnilinstances then
        pcall(function()
            nilInstances = getnilinstances()
        end)
    end
    
    for _, fish in ipairs(FishList) do
        if not FishEnabled[fish.Name] then continue end
        
        local found = nil
        
        if nilInstances then
            for _, obj in ipairs(nilInstances) do
                if obj.Name == fish.Name then
                    local ok, did = pcall(function() return obj:GetDebugId() end)
                    if ok and did == fish.DebugId then
                        found = obj
                        break
                    elseif not ok then
                        found = obj
                        break
                    end
                end
            end
        end
        
        if not found then
            local backpack = player:FindFirstChild("Backpack")
            if backpack then
                found = backpack:FindFirstChild(fish.Name)
            end
        end
        
        if not found then
            local char = player.Character
            if char then
                found = char:FindFirstChild(fish.Name)
            end
        end
        
        if found then
            table.insert(collected, {
                name = fish.Name,
                obj = found,
                amount = fish.Amount,
            })
        end
    end
    
    return collected
end

-- ═══ بيع الأسماك حسب العدد المحدد ═══
local function SellAllFish()
    local ev = getSellEvent()
    if not ev then return 0 end
    
    local fishes = CollectAllFish()
    if #fishes == 0 then return 0 end
    
    local sold = 0
    
    for _, item in ipairs(fishes) do
        -- نرسل السمكة بعدد المرات المحدد
        for i = 1, item.amount do
            pcall(function()
                ev:FireServer("SELL_ITEM", "tool", item.obj)
            end)
            sold = sold + 1
            sellCount = sellCount + 1
            
            -- تأخير بسيط بين كل إرسال لتجنب lag
            if FishSellConfig.Delay > 0 then
                task.wait(FishSellConfig.Delay)
            end
        end
    end
    
    return sold
end

-- ═══ الإيقاف ═══
local function StopSell()
    FishSellConfig.Enabled = false
    if sellThread then
        pcall(function() task.cancel(sellThread) end)
        sellThread = nil
    end
    A:Notify({
        Title = "بيع الاسماك",
        Content = "تم الإيقاف | العمليات: " .. sellCount,
        Duration = 2
    })
end

-- ═══ البدء ═══
local function StartSell()
    local ev = getSellEvent()
    if not ev then
        A:Notify({
            Title = "خطأ",
            Content = "ما لقيت SellShopEvent",
            Duration = 3
        })
        return
    end
    
    FishSellConfig.Enabled = true
    sellCount = 0
    
    sellThread = task.spawn(function()
        while FishSellConfig.Enabled do
            SellAllFish()
            task.wait(0.5)
        end
    end)
    
    A:Notify({
        Title = "بيع الاسماك",
        Content = "اشتغل",
        Duration = 2
    })
end
